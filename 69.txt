// === 根据关键词识别数据库类型 ===
detectDatabaseType = func(keyword) {
    keywordLower = str.ToLower(keyword)
    
    if str.Contains(keywordLower, "mysql") || str.Contains(keywordLower, "sqlstate[") {
        return "MySQL"
    } else if str.Contains(keywordLower, "postgresql") || str.Contains(keywordLower, "pg_") {
        return "PostgreSQL"
    } else if str.Contains(keywordLower, "microsoft sql") || str.Contains(keywordLower, "msg 102") {
        return "MSSQL"
    } else if str.Contains(keywordLower, "ora-") {
        return "Oracle"
    } else if str.Contains(keywordLower, "sqlite") {
        return "SQLite"
    } else if str.Contains(keywordLower, "access") || str.Contains(keywordLower, "odbc") {
        return "Access/ODBC"
    }
    
    return "Unknown"
}

// === 根据payload识别数据库类型 ===
detectDatabaseTypeFromPayloads = func(payloads) {
    for _, payload = range payloads {
        if payload == nil {
            continue
        }
        // 兼容对象payload，提取value
        payloadStr = ""
        if type(payload) == "string" {
            payloadStr = payload
        } else {
            payloadStr = payload["value"]
        }
        payloadLower = str.ToLower(payloadStr)
        
        if str.Contains(payloadLower, "sleep") {
            return "MySQL"
        } else if str.Contains(payloadLower, "pg_sleep") {
            return "PostgreSQL"
        } else if str.Contains(payloadLower, "waitfor delay") {
            return "MSSQL"
        } else if str.Contains(payloadLower, "dbms_pipe") {
            return "Oracle"
        }
    }
    
    return "Unknown"
}


// === 创建带代理的 fuzzer（统一封装）===
createFuzzerWithProxy = func(req, isHttps, proxy) {
    opts = [fuzz.https(isHttps)]
    if proxy != nil && proxy != "" {
        opts = append(opts, fuzz.proxy(proxy))
    }
    // 显式返回两个值：fuzzer 和 error
    fuzzer, err = fuzz.HTTPRequest(req, opts...)
    return fuzzer, err
}

// === 提取 payload 信息：返回 (value, mode) ===
extractPayloadInfo = func(item) {
    // 首先检查 item 是否为 nil，避免后续访问 panic
    if item == nil {
        return "", "append"
    }

    switch type(item) {
    case "string":
        // 字符串直接作为 value，mode 默认为 append
        return item, "append"
    case "map":
        // 显式赋值给临时变量，帮助静态分析器识别类型
        m = item
        value = m["value"]
        if value == nil {
            value = ""
        }
        mode = m["mode"]
        if mode == nil {
            mode = "append"
        }
        return value, mode
    default:
        // 其他类型（数字、布尔等）作为 fallback
        return "", "append"
    }
}

// === 检测响应是否为 WAF 拦截页面 ===
isWAFResponse = func(rawBody) {
    if rawBody == nil || rawBody == "" {
        return false
    }
    // 仅对响应体做一次小写转换
    lowerBody := str.ToLower(rawBody)
    // 直接使用启动时预计算的小写关键词切片
    for _, lowerKw := range lowerWafKeywords {
        if str.Contains(lowerBody, lowerKw) {
            println(sprintf("WAF 特征命中: %s", lowerKw))
            return true
        }
    }
    return false
}

// === 动态内容清洗：正则替换动态值 ===
cleanDynamicContent = func(rawBody) {
    if rawBody == nil || rawBody == "" {
        return rawBody
    }
    cleaned := rawBody
    // 直接使用启动时预编译的全局正则，无运行时编译开销
    for i, compiledRe := range compiledCleanPatterns {
        cleaned = compiledRe.ReplaceAllString(cleaned, cleanPlaceholders[i])
    }
    return cleaned
}




// === 工具函数：解析黑名单为 map ===
parseDomainListToMap = func(listStr) {
    m = make(map[string]bool)
    if listStr == nil || listStr == "" { 
        return m 
    }
    items = str.Split(listStr, ",")
    for i = 0; i < len(items); i++ {
        item = items[i]
        if item == nil {
            continue
        }
        itemStr = sprintf("%v", item)
        if itemStr == "" {
            continue
        }
        d = str.TrimSpace(itemStr)
        if d == "" {
            continue
        }
        d = str.ToLower(d)
        if str.HasPrefix(d, ".") {
            d = str.TrimPrefix(d, ".")
        }
        if str.HasSuffix(d, ".") {
            d = str.TrimSuffix(d, ".")
        }
        if d != "" {
            m[d] = true
        }
    }
    return m
}

    if listStr == nil || listStr == "" { 
        return m 
    }
    items = str.Split(listStr, ",")
    for i = 0; i < len(items); i++ {
        item = items[i]
        if item == nil {
            continue
        }
        itemStr = sprintf("%v", item)
        if itemStr == "" {
            continue
        }
        path = str.TrimSpace(itemStr)
        if path == "" {
            continue
        }
        path = str.ToLower(path)
        if path != "" {
            m[path] = true
        }
    }
    return m
}

// === 新增：白名单解析函数 ===
parseWhiteListToMap = func(listStr) {
    m = make(map[string]bool)
    if listStr == nil || listStr == "" { 
        return m 
    }
    items = str.Split(listStr, "\n")
    for i = 0; i < len(items); i++ {
        item = items[i]
        if item == nil {
            continue
        }
        itemStr = sprintf("%v", item)
        if itemStr == "" {
            continue
        }
        d = str.TrimSpace(itemStr)
        if d == "" {
            continue
        }
        if str.HasPrefix(d, "*") {
            d = str.TrimPrefix(d, "*")
        }
        d = str.ToLower(d)
        if d != "" {
            m[d] = true
        }
    }
    return m
}

// === 新增：通配符匹配函数 ===
matchGlobPattern = func(host, pattern) {
    if str.HasSuffix(host, pattern) {
        return true
    }
    if host == pattern {
        return true
    }
    return false
}

// === 获取字符串子串的辅助函数 ===
substring = func(s, start, end) {
    if s == nil || start < 0 || end > len(s) || start >= end {
        return ""
    }
    result = ""
    for i = start; i < end; i++ {
        result = result + string(s[i])
    }
    return result
}

// === Ghost Bits 转换函数 ===
toGhostBits = func(s) {
    result = ""
    for i = 0; i < len(s); i++ {
        b = s[i]                    // 字节值 (0~255)
        if b >= 0x20 && b <= 0x7E { // 可打印 ASCII
            key = sprintf("%d", b)
            ghostChar = ghostMap[key]
            if ghostChar != "" {
                result = result + ghostChar
            } else {
                result = result + string(b)
            }
        } else {
            // 非可打印 ASCII 或高位字节，原样保留
            result = result + string(b)
        }
    }
    return result
}

                            println(sprintf("   [%d] 跳过无效检测组: %v", idx+1, item))
                        }
                    }
                } else if type(arr) == "map" {
                    println("整体内容解析为单个 JSON 对象")
                    if arr["name"] != nil && arr["payloads"] != nil && arr["conditions"] != nil {
                        detectionGroups = append(detectionGroups, arr)
                        loadedGroups++
                        println(sprintf("加载检测组: %s", arr["name"]))
                    } else {
                        println("跳过无效对象")
                    }
                } else {
                    println("整体内容不是 JSON 数组或对象，切换到按行解析模式")
                    // 整体解析不成功，进入按行解析（密码字典模式）
                    for _, line = range rawLines {
                        line = str.TrimSpace(line)
                        if line == "" { continue }
                        try {
                            parsed = json.loads(line)
                            if type(parsed) == "array" {
                                // 这一行是一个 JSON 数组，遍历其中的每个元素
                                for _, item = range parsed {
                                    if type(item) == "map" && item["name"] != nil && item["payloads"] != nil && item["conditions"] != nil {
                                        detectionGroups = append(detectionGroups, item)
                                        loadedGroups++
                                        println(sprintf("   从行数组加载检测组: %s", item["name"]))
                                    } else {
                                        println(sprintf("   跳过行数组中无效元素: %v", item))
                                    }
                                }
                            } else if type(parsed) == "map" {
                                // 这一行是单个检测组对象
                                if parsed["name"] != nil && parsed["payloads"] != nil && parsed["conditions"] != nil {
                                    detectionGroups = append(detectionGroups, parsed)
                                    loadedGroups++
                                    println(sprintf("   从行对象加载检测组: %s", parsed["name"]))
                                } else {
                                    println(sprintf("   跳过无效对象: %v", parsed))
                                }
                            } else {
                                println(sprintf("   跳过非数组/非对象行: %s", line))
                            }
                        } catch parseErr {
                            println(sprintf("   跳过无法解析的行: %s", line))
                        }
                    }
                }
            } catch parseErr {
                println(sprintf("❌ 整体 JSON 解析失败: %v，切换到按行解析模式", parseErr))
                // 整体解析异常，进入按行解析
                for _, line = range rawLines {
                    line = str.TrimSpace(line)
                    if line == "" { continue }
                    try {
                        parsed = json.loads(line)
                        if type(parsed) == "array" {
                            for _, item = range parsed {
                                if type(item) == "map" && item["name"] != nil && item["payloads"] != nil && item["conditions"] != nil {
                                    detectionGroups = append(detectionGroups, item)
                                    loadedGroups++
                                    println(sprintf("   从行数组加载检测组: %s", item["name"]))
                                }
                            }
                        } else if type(parsed) == "map" {
                            if parsed["name"] != nil && parsed["payloads"] != nil && parsed["conditions"] != nil {
                                detectionGroups = append(detectionGroups, parsed)
                                loadedGroups++
                                println(sprintf("   从行对象加载检测组: %s", parsed["name"]))
                            }
                        }
                    } catch parseErr2 {
                        println(sprintf("   跳过无法解析的行: %s", line))
                    }
                }
            }
        }

    // === 添加阈值计算函数===
    calculateThreshold = func(condition, group, factor, baselineSim) {
        baseThreshold = condition["baseThreshold"]
        if baseThreshold == nil {
            baseThreshold = group["baseThreshold"]
        }
        if baseThreshold == nil {
            baseThreshold = 0.95
        }
        baseFloat := float(baseThreshold)

        groupType = group["type"]
        typeFactor = 1.0
        if groupType == "time" {
            typeFactor = 0.1
        } else if groupType == "error" {
            typeFactor = 0.5
        } else if groupType == "union" {
            typeFactor = 0.8
        } else if groupType == "boolean" {
            typeFactor = 1.0
        } else if groupType == "similarity" {
            typeFactor = 1.0
        }

        result := baseFloat * factor * typeFactor

        // 基线相似度校正：页面本身波动大时放宽阈值
        if baselineSim > 0.5 {
            correctFactor := baselineSim / 0.98
            if correctFactor > 1.0 {
                correctFactor = 1.0
            }
            result = result * correctFactor
        }

        if result > 1.0 {
            result = 1.0
        } else if result < 0.0 {
            result = 0.0
        }
        return result
    }

                    }
                }
            } else {

            }

    // === 交换插入模式处理 ===

        println("⚙️ 已启用「交换插入模式」，将为每个检测组扩展互换模式的 payload")
        for idx, group = range detectionGroups {
            originalPayloads = group["payloads"]
            if originalPayloads == nil || len(originalPayloads) == 0 {
                continue
            }
            newPayloads = []
            // 用于去重：记录已存在的 (value, mode) 组合
            seen = make(map[string]bool)
            for _, item = range originalPayloads {
                value, mode = extractPayloadInfo(item)
                key = sprintf("%s|%s", value, mode)
                if seen[key] {
                    continue
                }
                seen[key] = true
                // 保留原 payload
                newPayloads = append(newPayloads, item)
                // 生成互换模式的 payload
                swappedMode = "append"
                if mode == "append" {
                    swappedMode = "replace"
                } else if mode == "replace" {
                    swappedMode = "append"
                }
                swappedKey = sprintf("%s|%s", value, swappedMode)
                if !seen[swappedKey] {
                    if swappedMode == "append" {
                        // append 模式可以直接用字符串表示
                        newPayloads = append(newPayloads, value)
                    } else {
                        newPayloads = append(newPayloads, {"value": value, "mode": swappedMode})
                    }
                    seen[swappedKey] = true
                }
            }
            detectionGroups[idx]["payloads"] = newPayloads
            println(sprintf("  检测组「%s」: payload 数量从 %d 扩展至 %d", group["name"], len(originalPayloads), len(newPayloads)))
        }
    }




    // ========== 新增：Ghost Bits 编码模式 ==========

        println("👻 已启用「Ghost Bits 编码模式」，将为每个检测组生成汉字幽灵载荷...")
        for idx, group = range detectionGroups {
            originalPayloads = group["payloads"]
            if originalPayloads == nil || len(originalPayloads) == 0 {
                continue
            }
            newPayloads = []
            seen = make(map[string]bool)
            for _, item = range originalPayloads {
                // 保留原始 payload
                value, mode = extractPayloadInfo(item)
                key = sprintf("%s|%s", value, mode)
                if !seen[key] {
                    newPayloads = append(newPayloads, item)
                    seen[key] = true
                }

                // 生成 Ghost Bits 版本
                ghostValue = toGhostBits(value)
                if ghostValue != "" && ghostValue != value {
                    ghostKey = sprintf("%s|%s", ghostValue, mode)
                    if !seen[ghostKey] {
                        if mode == "append" {
                            // 字符串模式
                            newPayloads = append(newPayloads, ghostValue)
                        } else {
                            // 对象模式（Yak 原生风格，不用 map[string]interface{}）
                            newPayloads = append(newPayloads, {"value": ghostValue, "mode": mode})
                        }
                        seen[ghostKey] = true
                    }
                }
            }
            detectionGroups[idx]["payloads"] = newPayloads
            println(sprintf("  检测组「%s」: Ghost Bits 扩展后 payload 数量: %d", group["name"], len(newPayloads)))
        }
    }




    // === 创建并发控制 ===
    wg = sync.NewSizedWaitGroup(maxConcurrent)

    // === 对每个参数进行并发测试 ===

        if currentParam == nil {
            println("警告: 遇到空参数对象，跳过")
            continue
        }
        
        paramName = currentParam.Name()
        paramNameLower = str.ToLower(paramName)
        if paramNameLower in paramBlacklistMap {
            println(sprintf("参数名黑名单命中，跳过参数: %s", paramName))
            continue
        }
        
        if notTestCookie && currentParam.IsCookieParams() {
            println(sprintf("跳过Cookie参数: %s", paramName))
            continue
        }
        
        originalValue = currentParam.Value()
        if type(originalValue) != "string" {
            originalValue = sprintf("%v", originalValue)
        }
        originalValueLower = str.ToLower(originalValue)
        if originalValueLower in paramValueBlacklistMap {
            println(sprintf("参数值黑名单命中，跳过参数: %s (值: %s)", paramName, originalValue))
            continue
        }
        
        if len(originalValue) < minParamValueLength {
            println(sprintf("参数值长度小于最小长度，跳过参数: %s (长度: %d)", paramName, len(originalValue)))
            continue
        }
        
        wg.Add()
        
        // 启动并发 goroutine

            defer wg.Done()

            paramName = param.Name()
            originalValue = param.Value()
            if type(originalValue) != "string" {
                originalValue = sprintf("%v", originalValue)
            }
            println(sprintf("正在测试参数: %s, 原始值: %s", paramName, originalValue))

            paramType = ""
            if param.IsGetParams() {
                paramType = "GET"
            } else if param.IsPostParams() {
                paramType = "POST"
            } else if param.IsCookieParams() {
                paramType = "COOKIE"
            } else {
                paramType = "OTHER"
            }

            // === MODIFIED START: 用于合并敏感信息泄露报告的对象（使用空对象字面量）===
            sensitiveAlerts = make(map[string]any)
            // === MODIFIED END ===

            // 初始化滑动时间窗口，填入全局原始请求的时间样本
            timeWindow := make([]int, 0, 10)
            for _, t := range timeSamples {
                if len(timeWindow) < 10 {
                    timeWindow = append(timeWindow, t)
                }
            }


            wafDetected = false
            wafReported = false
            for _, group = range detectionGroups {
                // 如果开启了 stop-on-waf 且已经检测到 WAF，则跳出循环，停止该参数所有后续检测组
                if stopOnWaf && wafDetected {
                    println(sprintf("参数 [%s] 因检测到 WAF 且 stop-on-waf 开启，停止后续检测组", paramName))
                    break
                }
                if group["type"] == "union" && !enableUnionDetection {
                    continue
                }

                if len(group["payloads"]) == 0 {
                    continue
                }

                println(sprintf("参数 [%s] 执行检测组: %s (%s)", paramName, group["name"], group["type"]))

                responses = make(map[string]string)
                similarities = make(map[string]float64)
                responseTimes = make(map[string]int)

                // FIX: 新增映射，用于存储每个payload对应的完整测试值
                fullPayloadMap = make(map[string]string)

                var fullPayload = ""

                // === 修改为支持 payload 级插入模式的请求发送方式 ===
                for _, payloadItem = range group["payloads"] {
                    // 提取 payload 字符串和模式
                    payloadStr, payloadMode = extractPayloadInfo(payloadItem)

                    // 根据模式构造完整测试参数值
                    if payloadMode == "replace" {
                        fullPayload = payloadStr
                    } else {
                        fullPayload = originalValue + payloadStr
                    }
                    // FIX: 存储payloadStr对应的完整值
                    fullPayloadMap[payloadStr] = fullPayload

                    println(sprintf("参数 [%s] 测试载荷: %s (模式: %s)", paramName, fullPayload, payloadMode))

                    testReqs, _ = param.Fuzz(fullPayload).Results()

                for _, testRequest = range testReqs {
                    // 随机延迟（如果启用）
                    if (randomDelayMax > 0 && randomDelayMax > randomDelayMin) {
                            delay := randn(randomDelayMin, randomDelayMax)
                            if (delay > 0) {
                                time.Sleep(float64(delay) / 1000.0)
                            }
                        }

                    // 使用 fuzz 发送测试请求，支持超时和重试
                    // testFuzzer, err := fuzz.HTTPRequest(testRequest, fuzz.https(isHttps))
                    // testFuzzer, err := fuzz.HTTPRequest(testRequest, 
                    // fuzz.https(isHttps),
                    // fuzz.proxy(proxy1)
                    // )
                    testFuzzer, err := createFuzzerWithProxy(testRequest, isHttps, proxy1)
                    if err != nil {
                        println(sprintf("创建测试请求的fuzzer失败: %v", err))
                        continue
}
                    
                    testResp, reqErr := executeWithRetry(testFuzzer, httpTimeout)
                    if reqErr != nil {
                        println(sprintf("测试请求失败（重试后）: %v, payload=%s", reqErr, payloadStr))
                        continue
                    }

                    // 基础响应信息
                    responseTime = testResp.ServerDurationMs
                    testBody = string(testResp.LowhttpResponse.GetBody())

                    // 获取响应状态码（从 LowhttpResponse 调用 GetStatusCode()）
                    statusCode = 0
                    if testResp.LowhttpResponse != nil {
                        statusCode = testResp.LowhttpResponse.GetStatusCode()
                    }

                    // 获取请求 URL（从 testRequest 对象获取完整 URL）
                    requestURL = ""
                    if testRequest != nil && testRequest.URL != nil {
                        requestURL = testRequest.URL.String()
                    } else {
                        requestURL = url // 备用，可能不含修改后的参数
                    }

                    // WAF 检测
                    wafBlocked = isWAFResponse(testBody)

                    // 初始化相似度和错误关键字
                    similarity = 0.0
                    matchedKeyword = ""

                    // 如果不是 WAF 拦截，计算相似度和检测错误关键字
                    cleanedTestBody := cleanDynamicContent(testBody)
                    if !wafBlocked {
                        similarity, simErr = str.CalcTextMaxSubStrStability(cleanedOriginalBody, cleanedTestBody)
                        if simErr != nil {
                            println(sprintf("计算相似度失败: %v, payload=%s", simErr, payloadStr))
                            similarity = 0.0
                        }

                        lowerTestBody = str.ToLower(testBody)
                        for _, keyword = range sensitiveKeywords {
                            if str.Contains(lowerTestBody, str.ToLower(keyword)) {
                                matchedKeyword = keyword
                                break
                            }
                        }
                    }
                    responses[payloadStr] = cleanedTestBody   // 存储清洗后的，供 payload 间相似度对比

                    // ========== 插入流量统计表格（如果启用）==========
                        seqLock.Lock()
                        rowSeq++
                        currentSeq := rowSeq
                        seqLock.Unlock()

                        tableLock.Lock()
                        yakit.TableData("SQL注入测试流量", {
                            "序号":           currentSeq,
                            "参数名":         paramName,
                            "原始值":         originalValue,
                            "测试payload":    payloadStr,
                            "完整测试值":     paramName + "=" + fullPayload,
                            "检测组":         group["name"],
                            "请求URL":        requestURL,
                            "响应状态码":     statusCode,
                            "响应大小(bytes)": len(testBody),
                            "响应时间(ms)":   responseTime,
                            "相似度(与原始)": similarity,
                            "是否WAF拦截":    wafBlocked ? "是" : "否"
                        })
                        tableLock.Unlock()

                    // 如果被 WAF 拦截，跳过后续处理（但已记录表格）
                    if wafBlocked {
                        // 如果尚未上报 WAF 风险，则上报一次
                        if !wafReported {
                            // 获取请求 URL
                            requestURL = ""
                            if testRequest != nil && testRequest.URL != nil {
                                requestURL = testRequest.URL.String()
                            } else {
                                requestURL = url
                            }

                            risk.NewRisk(
                                host,  // 主机名（外部闭包变量，可用）
                                risk.title(sprintf("WAF拦截: 参数 %s 被防火墙拦截 (%s)", paramName, group["name"])),
                                risk.severity("info"),
                                risk.parameter(paramName),
                                risk.payload(payloadStr),
                                risk.description(sprintf("请求被 Web 应用防火墙拦截，可能触发安全规则。URL: %s", requestURL)),
                                risk.solution("检查 payload 是否触发 WAF，或调整测试策略。"),
                                risk.typeVerbose("waf-intercept"),
                                risk.request(testResp.RawRequest),
                                risk.response(testResp.RawPacket),
                                risk.details({
                                    "detection_group": group["name"],
                                    "param_name": paramName,
                                    "param_type": paramType,
                                    "original_value": originalValue,
                                    "payload": payloadStr,
                                    "response_time_ms": responseTime,
                                    "waf_keyword_matched": "检测到WAF特征"
                                })
                            )
                            wafReported = true
                        }
                        wafDetected = true
                        println(sprintf("检测到 WAF 拦截页面，跳过 payload: %s", payloadStr))
                        break  // 跳出当前 payload 循环
                    }

                    // 记录响应数据供后续分析（原逻辑保留）
                    responseTimes[payloadStr] = responseTime
                    similarities[payloadStr] = similarity

                    // 更新滑动窗口（仅非时间组，且排除异常长响应）
                    if group["type"] != "time" && responseTime > 0 && responseTime < 30000 {
                        if len(timeWindow) >= 10 {
                            timeWindow = timeWindow[1:]   // 保持固定大小
                        }
                        timeWindow = append(timeWindow, responseTime)
                    }




                    // 敏感关键字收集（原逻辑）
                    lowerTestBody = str.ToLower(testBody)
                    for _, keyword = range sensitiveKeywords {
                        if str.Contains(lowerTestBody, str.ToLower(keyword)) {
                            alertKey = sprintf("%s:%s", paramName, keyword)
                            fullPayloadActual = fullPayloadMap[payloadStr]

                            keywordPos = str.Index(lowerTestBody, str.ToLower(keyword))
                            startPos = keywordPos - 40
                            if startPos < 0 { startPos = 0 }
                            endPos = keywordPos + len(keyword) + 40
                            if endPos > len(testBody) { endPos = len(testBody) }
                            contextStr = substring(testBody, startPos, endPos)

                            entry = sensitiveAlerts[alertKey]
                            if entry == nil {
                                entry = make(map[string]any)
                                entry["payloads"] = []
                                entry["contexts"] = []
                                entry["groups"]   = []
                                entry["sims"]     = []
                                entry["times"]    = []
                            }
                            entry["payloads"] = append(entry["payloads"], fullPayloadActual)
                            entry["contexts"] = append(entry["contexts"], contextStr)
                            entry["groups"]   = append(entry["groups"], group["name"])
                            entry["sims"]     = append(entry["sims"], similarity)
                            entry["times"]    = append(entry["times"], responseTime)
                            sensitiveAlerts[alertKey] = entry

                            break // 每个 payload 只记录第一个匹配的关键字
                        }
                    }

                    break // 每个载荷只测试第一个请求
                }
                }

                // 计算载荷之间的相似度
                for _, condition = range group["conditions"] {
                    if condition["compareWith"] != "original" {
                        comparePayload = condition["compareWith"]
                        targetPayload = condition["payload"]

                        if responses[comparePayload] != nil && responses[targetPayload] != nil {
                            similarity, simErr = str.CalcTextMaxSubStrStability(responses[comparePayload], responses[targetPayload])
                            if simErr == nil {
                                similarities[targetPayload + "_vs_" + comparePayload] = similarity
                                println(sprintf("参数 [%s] %s 与 %s 相似度: %f", paramName, targetPayload, comparePayload, similarity))
                            }
                        }
                    }
                }

                // 定义 conditionResults (用于风险详情)
                conditionResults = make(map[string]bool)

                // FIX: 新增数组，记录所有满足条件的payload
                satisfiedPayloads = []

                // ========== 修改开始：根据组自定义逻辑决定判定逻辑 ==========
                conditionResults = make(map[string]bool)
                satisfiedPayloads = []   // 记录所有满足条件的payload
                isVulnerable = false     // 默认无漏洞

                // 获取该组的判定逻辑，若未设置则根据 type 提供默认值
                logic = group["logic"]
                if logic == nil {
                    if group["type"] == "time" {
                        logic = "any"
                    } else {
                        logic = "all"
                    }
                }

                if logic == "any" {
                    // 任一条件满足即可
                    for _, condition = range group["conditions"] {
                        conditionPassed := false
                        payload = condition["payload"]
                        compareWith = condition["compareWith"]

                        currentThreshold = calculateThreshold(condition, group, similarityThresholdFactor, baselineSimilarity)

                        useDynamicThreshold := false
                        if group["type"] == "time" {

                            if condition["useDynamicThreshold"] == nil || condition["useDynamicThreshold"] != false {
                                useDynamicThreshold = true
                            }
                        } else if condition["useDynamicThreshold"] == true {
                            // 非 time 组可显式开启
                            useDynamicThreshold = true
                        }

                        // 2. 是否检查时间：任何情况只要开启动态阈值或组原本就启用了时间检查
                        shouldCheckTime := group["type"] == "time" || group["enableTimeCheck"] == true || useDynamicThreshold

                        thresholdOperator = condition["thresholdOperator"]
                        // 仅在时间组才读取时间阈值字段，否则使用无影响的默认值
                        timeThreshold = 0
                        timeThresholdOperator = "<"
                        if shouldCheckTime {
                            timeThreshold = condition["timeThreshold"]
                            timeThresholdOperator = condition["timeThresholdOperator"]
                        }

                                // 计算均值
                                sum := 0
                                for _, t := range timeWindow {
                                    sum += t
                                }
                                avg := float64(sum) / float64(len(timeWindow))
                                // 计算标准差
                                sumSq := 0.0
                                for _, t := range timeWindow {
                                    diffVal := float64(t) - avg
                                    sumSq += diffVal * diffVal
                                }
                                std := math.Sqrt(sumSq / float64(len(timeWindow)))

                                // 偏移量取 3倍标准差 与 3000 中的较大值
                                offset := std * 3
                                if offset < 3000 {
                                    offset = 3000
                                }
                                dynamicThreshold := int(avg + offset)
                                if dynamicThreshold < 4000 {
                                    dynamicThreshold = 4000
                                }
                                timeThreshold = dynamicThreshold
                            } else {
                                timeThreshold = 4000
                            }
                        }




                        similarityKey = payload
                        if compareWith != "original" {
                            similarityKey = payload + "_vs_" + compareWith
                        }

                        similarity = similarities[similarityKey]
                        if similarity == nil {
                            println(sprintf("警告: 未找到相似度值 for key: %s, 跳过此条件", similarityKey))
                            continue
                        }

                        responseTime = responseTimes[payload]
                        if responseTime == nil {
                            println(sprintf("警告: 未找到响应时间 for payload: %s, 跳过此条件", payload))
                            continue
                        }

                        responseTimeInt = int(responseTime)

                        similarityMet = false
                        if thresholdOperator == "<" {
                            similarityMet = similarity < currentThreshold
                        } else if thresholdOperator == ">" {
                            similarityMet = similarity > currentThreshold
                        } else if thresholdOperator == "=" {
                            similarityMet = similarity == currentThreshold
                        }

                        timeMet = true   // 默认通过，非时间组直接 true
                        if shouldCheckTime {
                            if timeThresholdOperator == "<" {
                                timeMet = responseTimeInt < timeThreshold
                            } else if timeThresholdOperator == ">" {
                                timeMet = responseTimeInt > timeThreshold
                            } else if timeThresholdOperator == "=" {
                                timeMet = responseTimeInt == timeThreshold
                            }
                        }

                        // 时间组忽略相似度，仅用时间判断
                        if group["type"] == "time" {
                            conditionPassed = timeMet
                        } else {
                            conditionPassed = similarityMet && timeMet
                        }
                        conditionResults[payload + "_similarity"] = similarityMet
                        conditionResults[payload + "_time"] = timeMet

                        println(sprintf("条件检查: payload=%s, similarity=%f, currentThreshold=%f, similarityMet=%v, responseTime=%d, timeMet=%v, conditionPassed=%v",
                            payload, similarity, currentThreshold, similarityMet, responseTimeInt, timeMet, conditionPassed))

                        if conditionPassed {
                            satisfiedPayloads = append(satisfiedPayloads, payload)
                            isVulnerable = true
                            break
                        }
                    }
                } else { // logic == "all"
                    // 所有条件必须满足
                    isVulnerable = true
                    for _, condition = range group["conditions"] {
                        conditionPassed := false
                        payload = condition["payload"]
                        compareWith = condition["compareWith"]
                        
                        // 动态计算当前阈值
                        currentThreshold = calculateThreshold(condition, group, similarityThresholdFactor, baselineSimilarity)
                        
                        // 1. 先判断本条件是否使用动态阈值
                        useDynamicThreshold := false
                        if group["type"] == "time" {
                            // time 组默认使用动态阈值，除非显式关闭
                            if condition["useDynamicThreshold"] == nil || condition["useDynamicThreshold"] != false {
                                useDynamicThreshold = true
                            }
                        } else if condition["useDynamicThreshold"] == true {
                            // 非 time 组可显式开启
                            useDynamicThreshold = true
                        }

                        // 2. 是否检查时间：任何情况只要开启动态阈值或组原本就启用了时间检查
                        shouldCheckTime := group["type"] == "time" || group["enableTimeCheck"] == true || useDynamicThreshold

                        thresholdOperator = condition["thresholdOperator"]
                        // 仅在时间组才读取时间阈值字段，否则使用无影响的默认值
                        timeThreshold = 0
                        timeThresholdOperator = "<"
                        if shouldCheckTime {
                            timeThreshold = condition["timeThreshold"]
                            timeThresholdOperator = condition["timeThresholdOperator"]
                        }




                        similarityKey = payload
                        if compareWith != "original" {
                            similarityKey = payload + "_vs_" + compareWith
                        }

                        similarity = similarities[similarityKey]
                        if similarity == nil {
                            println(sprintf("警告: 未找到相似度值 for key: %s, 此条件不满足", similarityKey))
                            isVulnerable = false
                            break
                        }

                        responseTime = responseTimes[payload]
                        if responseTime == nil {
                            println(sprintf("警告: 未找到响应时间 for payload: %s, 此条件不满足", payload))
                            isVulnerable = false
                            break
                        }

                        responseTimeInt = int(responseTime)

                        similarityMet = false
                        if thresholdOperator == "<" {
                            similarityMet = similarity < currentThreshold
                        } else if thresholdOperator == ">" {
                            similarityMet = similarity > currentThreshold
                        } else if thresholdOperator == "=" {
                            similarityMet = similarity == currentThreshold
                        }

                        timeMet = true
                        if shouldCheckTime {
                            if timeThresholdOperator == "<" {
                                timeMet = responseTimeInt < timeThreshold
                            } else if timeThresholdOperator == ">" {
                                timeMet = responseTimeInt > timeThreshold
                            } else if timeThresholdOperator == "=" {
                                timeMet = responseTimeInt == timeThreshold
                            }
                        }

                        // 时间组忽略相似度，仅用时间判断
                        if group["type"] == "time" {
                            conditionPassed = timeMet
                        } else {
                            conditionPassed = similarityMet && timeMet
                        }
                        conditionResults[payload + "_similarity"] = similarityMet
                        conditionResults[payload + "_time"] = timeMet

                        println(sprintf("条件检查: payload=%s, similarity=%f, currentThreshold=%f, similarityMet=%v, responseTime=%d, timeMet=%v, conditionPassed=%v",
                            payload, similarity, currentThreshold, similarityMet, responseTimeInt, timeMet, conditionPassed))

                        if conditionPassed {
                            satisfiedPayloads = append(satisfiedPayloads, payload)
                        }

                        if !conditionPassed {
                            isVulnerable = false
                            break
                        }
                    }

                    // 如果 conditions 为空，则无漏洞
                    if len(group["conditions"]) == 0 {
                        isVulnerable = false
                    }
                }
                // ========== 修改结束 ==========


                if isVulnerable {

                        vulnerablePayload = satisfiedPayloads[0]   // 取第一个满足条件的payload
                    } else {

                        firstItem = group["payloads"][0]
                        if type(firstItem) == "string" {
                            vulnerablePayload = firstItem
                        } else {
                            vulnerablePayload = firstItem["value"]
                        }
                    }
                    vulnerableFullPayload = fullPayloadMap[vulnerablePayload]
                    if vulnerableFullPayload == "" {
                        vulnerableFullPayload = vulnerablePayload // 最后的保底
                    }

                    payloadMetrics = []

                        similarity = similarities[payloadStr]
                        if similarity == nil {
                            similarity = 0.0
                        }
                        responseTime = responseTimes[payloadStr]
                        if responseTime == nil {
                            responseTime = 0
                        }
                        
                        // 检查该 payload 是否满足条件（即被判定为触发漏洞的 payload 之一）
                        isSatisfied = false
                        for _, sp = range satisfiedPayloads {
                            if sp == payloadStr {
                                isSatisfied = true
                                break
                            }
                        }
                        
                        metric = {
                            "payload": payloadStr,
                            "mode": payloadMode,
                            "full_payload": fullPayload,
                            "similarity": similarity,
                            "response_time_ms": responseTime,
                            "is_satisfied": isSatisfied
                        }
                        payloadMetrics = append(payloadMetrics, metric)
                    }

                    // 重新发送请求以获取证据
                    testReqs, _ = param.Fuzz(vulnerableFullPayload).Results()

                    if len(testReqs) == 0 {
                        println(sprintf("构造报告请求失败: 结果为空"))
                        continue

                    if reqErr != nil {
                        println(sprintf("发送报告请求失败: %v", reqErr))
                        continue
                    }
                    if testResp == nil {
                        println("报告请求响应为空")
                        continue
                    }

                    reqIns, err1 = poc.ParseBytesToHTTPRequest(testResp.RawRequest)
                    if err1 != nil {
                        println(sprintf("解析请求失败: %v", err1))
                        continue
                    }

                    rspIns, err2 = poc.ParseBytesToHTTPResponse(testResp.RawPacket)
                    if err2 != nil {
                        println(sprintf("解析响应失败: %v", err2))
                        continue
                    }

                    riskDetails = {
                        "detection_group": group["name"],
                        "group_type": group["type"],
                        "original_value": originalValue,
                        "similarities": similarities,
                        "response_times": responseTimes,
                        "condition_results": conditionResults,
                        "satisfied_payloads": satisfiedPayloads, // FIX: 添加满足条件的payload列表
                        "param_type": paramType,
                        "param_position": param.PositionVerbose(),
                        "detection_method": "multi_payload_analysis",
                        "similarity_threshold_factor": similarityThresholdFactor,
                        "threshold_adjusted": true,
                        "database_type_hint": detectDatabaseTypeFromPayloads(group["payloads"]),
                        "payload_metrics": payloadMetrics
                    }

                    risk.NewRisk(
                        reqIns.URL.Host,
                        risk.title(sprintf("SQL注入 (%s): %s", group["name"], reqIns.URL.String())),

                        risk.parameter(paramName),

                        risk.type("sqlinjection"),
                        risk.response(testResp.RawPacket),
                        risk.details(riskDetails)
                    )

                    println(sprintf("已报告漏洞: 参数 %s, 载荷 %s", paramName, vulnerableFullPayload))
                } else {
                    println(sprintf("参数 [%s] 检测组 %s 未检测到漏洞", paramName, group["name"]))
                }
            }

            // === MODIFIED START: 合并上报敏感信息泄露 ===
            for alertKey, entry = range sensitiveAlerts {
                // 从 key 中分离出参数名和关键字
                parts = str.Split(alertKey, ":")
                if len(parts) != 2 { continue }
                paramNameKey = parts[0]
                keyword = parts[1]

                // 合并 payload 显示（用 " | " 连接）
                mergedPayloads = str.Join(entry["payloads"], " | ")

                // 上下文取第一个
                firstContext = entry["contexts"][0]

                // 构建合并后的详情
                mergedDetails = {
                    "merged_payloads":    entry["payloads"],        // 原始列表
                    "merged_contexts":    entry["contexts"],
                    "trigger_groups":     entry["groups"],
                    "similarities":       entry["sims"],
                    "response_times_ms":  entry["times"],
                    "param_type":         paramType,
                    "param_position":     param.PositionVerbose(),
                    "database_type":      detectDatabaseType(keyword),
                }

                // 上报合并后的风险
                risk.NewRisk(
                    host,
                    risk.title(sprintf("敏感信息泄露: 检测到SQL关键字 '%s'", keyword)),
                    risk.severity("info"),
                    risk.parameter(paramNameKey),
                    risk.payload(mergedPayloads),   // 合并后的 payload 字符串

                println(sprintf("已合并上报敏感信息: 参数 %s, 关键字 %s, 共 %d 个 payload", paramNameKey, keyword, len(entry["payloads"])))
            }
            // === MODIFIED END ===

            // === Header注入检测（保持不变）===
            if enableHeaderInjection {
                testHeaders = ["Referer", "X-Forwarded-For", "X-Real-IP", "X-Forwarded-Host", "User-Agent"]

                            } else if headerName == "X-Forwarded-For" || headerName == "X-Real-IP" {
                                randomIP = sprintf("%d.%d.%d.%d", randn(1, 254), randn(1, 254), randn(1, 254), randn(1, 254))
                                testValue = sprintf("%s%s", randomIP, payload)
                            } else {
                                testValue = payload
                            }
                        }

                        if randomDelayMax > 0 {
